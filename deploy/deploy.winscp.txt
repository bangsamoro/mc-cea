# WinSCP script - publish the MC-CEA static site from C to B (B-HOST).
#
# Run from C (deploy-mc-cea.bat does this for you):
#   "C:\Program Files (x86)\WinSCP\WinSCP.com" /ini=nul ^
#       /script="deploy\deploy.winscp.txt" /parameter // "C:\Users\DELL\MCCEA"
#
# %1% = the local folder to publish (the folder that contains index.html).

option batch abort
option confirm off
option transfer binary

# --- connect --------------------------------------------------------------
# Saved WinSCP session for B. Name it exactly "B" once in the WinSCP GUI
# (New session -> SFTP, host B-HOST, port 22, your SSH user, save,
# log in once so the host key is cached). Then this line just works.
open B

# Explicit alternative - replace the "open B" line above with this one
# (connect once in the GUI first, or paste B's real fingerprint):
# open sftp://root@B-HOST/ -hostkey="ssh-ed25519 256 xx:xx:..."

# --- target folder --------------------------------------------------------
# Created automatically the first time; harmless if it already exists.
option batch continue
mkdir /var/www/html/mc-cea
option batch abort

# --- push -----------------------------------------------------------------
# Mirror %1% into the site folder on B; changed files are overwritten.
#
# NOTE: "-delete" is deliberately NOT used. It would delete files on B that
# are not in the local folder - you do not want that on a first deploy, and
# there is nothing on B to prune yet. Add it if you later want the remote
# folder to match the local one exactly:
#     synchronize remote -delete -criteria=time -- ...
#
# The filemask keeps dev-only material off the server:
#   .git*, *.md, deploy/, preview/, assets/ (empty leftover), *.log
#   ... no business on the web
synchronize remote -criteria=time -- "%1%" "/var/www/html/mc-cea" -filemask="| .git/; .gitignore; .gitattributes; *.md; deploy/; preview/; assets/; *.log"

exit
