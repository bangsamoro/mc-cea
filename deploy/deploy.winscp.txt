# WinSCP script - publish the MC-CEA portal from C to B.
#
# Normally run by deploy-mc-cea.bat, which passes two parameters:
#   %1%  the local folder to publish (the one holding index.html)
#   %2%  how to reach B - EITHER the name of a saved WinSCP session
#        (default "B"), OR a full connection URL such as
#        sftp://root@B-HOST/
#
# Called by hand:
#   "C:\Program Files (x86)\WinSCP\WinSCP.com" ^
#       /script="deploy\deploy.winscp.txt" /parameter // "C:\Users\DELL\MCCEA" "B"
#
# NOTE: do NOT add /ini=nul. With it WinSCP ignores saved sessions and the
#       connection string below degrades into a hostname lookup.

option batch abort
option confirm off
option transfer binary

# --- connect --------------------------------------------------------------
# %2% is either a saved session name or a URL - WinSCP accepts both.
open "%2%"

# If you ever want it hard-coded, this is the form (connect once first so the
# host key is cached, or paste B's real fingerprint):
# open sftp://root@B-HOST/ -hostkey="ssh-ed25519 256 xx:xx:..."

# --- push -----------------------------------------------------------------
# Mirror %1% into B's docroot, so the portal *is* the site home page.
# The docroot already exists, so there is no mkdir step.
#
# NOTE: "-delete" is deliberately NOT used. It would remove files on B that
# are not in the local folder - which at the docroot means the other
# applications (ifr/, tag/, phpmyadmin/). Add it only if you know exactly
# what else lives there:
#     synchronize remote -delete -criteria=time -- ...
#
# Filemask - what is NOT uploaded:
#   .htaccess ..... the docroot may already have one, and ours would cascade
#                   over ifr/ and tag/. Nothing sensitive is uploaded, so the
#                   portal does not need it.
#   .git*, *.md, deploy/, preview/, assets/, *.log ... dev-only material
synchronize remote -criteria=time -- "%1%" "/var/www/html" -filemask="| .htaccess; .git/; .gitignore; .gitattributes; *.md; deploy/; preview/; assets/; *.log"

exit
